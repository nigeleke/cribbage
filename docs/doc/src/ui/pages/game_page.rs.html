12#[component]
13pub fn GamePage(game_id: GameIdDTO) -> Element {
14    let user_id = use_context::<Signal<UserIdDTO>>();
15    let game_id = provide_context(game_id);
16
17    let mut game = use_signal(Option::<UserGameDTO>::default);
18
19    let mut game_stream = use_action(move || async move {
20        let mut stream = api::stream::user_game_stream(*user_id.read(), game_id).await?;
21        while let Some(Ok(updated_game)) = stream.next().await {
22            game.set(Some(updated_game));
23        }
24        dioxus::Ok(())
25    });
26
27    let get_game_result = use_resource(move || async move {
28        let current_game = api::queries::get_game(*user_id.read(), game_id).await?;
29        game_stream.call();
30        game.set(Some(current_game));
31        dioxus::Ok(())
32    });
33
34    use_effect(move || {
35        if let Some(Err(error)) = get_game_result.result() {
36            Toast::server_error("get game", error.to_string());
37        }
38    });
39
40    rsx! {
41        document::Stylesheet { href: asset!("/assets/css/game_page.css") }
42        div {
43            class: "game-page",
44            if let Some(game) = game() {
45                ActiveGame { game }
46            } else {
47                div { "Loading..." }
48            }
49        }
50    }
51}

53#[component]
54fn ActiveGame(game: ReadSignal<UserGameDTO>) -> Element {
55    provide_context(game);
56
57    match game().phase {
58        PhaseDTO::InLobby => rsx! { InLobby {} },
59        PhaseDTO::CuttingForDeal => rsx! { CuttingForDeal {} },
60        PhaseDTO::Discarding => rsx! { Discarding {} },
61        PhaseDTO::Playing => rsx! { Playing {} },
62        PhaseDTO::ScoringPone => rsx! { ScoringPone {} },
63        PhaseDTO::ScoringDealer => rsx! { ScoringDealer {} },
64        PhaseDTO::ScoringCrib => rsx! { ScoringCrib {} },
65        PhaseDTO::Finished => rsx! { Finished {} },
66    }
67}
