1use api::dto::{CardIdDTO, GameIdDTO, PlayActionDTO, PlayerDTO, PlaysDTO, UserIdDTO};
2use dioxus::prelude::*;
3
4use crate::{
5    components::{WaitingForOpponent, button::Button},
6    toast::Toast,

9#[component]
10pub fn PlayAction() -> Element {
11    let user_id = use_context::<Signal<UserIdDTO>>();
12    let game_id = use_context::<GameIdDTO>();
13
14    let selected_cards = use_context::<ReadSignal<Vec<CardIdDTO>>>();
15    let selected_count = selected_cards.read().len();
16
17    let plays = use_context::<ReadSignal<PlaysDTO>>();
18
19    let next_action = plays().next_action;
20    let user_play_turn = next_action == PlayActionDTO::Play(PlayerDTO::User);
21
22    let can_play = if user_play_turn && let Some(cid) = selected_cards.first() {
23        selected_count == 1 && plays().legal_plays.contains(&cid)
24    } else {
25        false
26    };
27
28    let mut play_action = use_action(move || async move {
29        if let Some(cid) = selected_cards().first() {
30            let result = api::action::play_card(*user_id.read(), game_id, cid.clone()).await;
31            match result {
32                Ok(_) => (),
33                Err(ref error) => {
34                    Toast::command_error("Play", error.to_string());
35                }
36            }
37            result
38        } else {
39            Ok(())
40        }
41    });
42
43    let on_play = move |_| play_action.call();
44
45    rsx! {
46        if let PlayActionDTO::Play(player) = next_action {
47            if player == PlayerDTO::User {
48                Button {
49                    onclick: on_play,
50                    disabled: !can_play,
51                    "Play"
52                }
53            } else {
54                WaitingForOpponent {}
55            }
56        }
57
58    }
59}
